# AKNexus Framework 关键代码基座

本文档定义各层的核心代码结构，是骨架维护的参照标准。

---

## 一、整体串联关系

```
Program.cs (启动)
    │
    ├─→ WebApplicationExtensions (Framework.Web)
    │       │
    │       ├─→ AddInject() [Furion]
    │       └─→ UseInject() [Furion]
    │
    ├─→ HubDbContext (Infrastructure)
    │       │
    │       ├─→ 继承 BaseDbContext
    │       └─→ 配置实体映射
    │
    └─→ 注册服务
            │
            ├─→ IRepository<> → EfRepositoryBase (Infrastructure)
            ├─→ IUnitOfWork → UnitOfWork (Infrastructure)
            └─→ IUserService → UserService (Application)
```

namespace AKNexus.Framework.Domain.Entities;

/// <summary>实体基类</summary>
public abstract class EntityBase : IEntity
{
    public Guid Id { get; protected set; } = Guid.NewGuid();
    
    protected EntityBase() { }
    protected EntityBase(Guid id) => Id = id;
    
    public override bool Equals(object? obj) => obj is EntityBase other && Id == other.Id;
    public override int GetHashCode() => Id.GetHashCode();
}

/// <summary>软删除实体基类</summary>
public abstract class SoftDeleteEntity : EntityBase, ISoftDelete
{
    public bool IsDeleted { get; set; }
    public DateTimeOffset? DeletedTime { get; set; }
}

/// <summary>完整审计实体基类</summary>
public abstract class FullAuditEntity : SoftDeleteEntity, IAuditable
{
    public Guid? CreatedBy { get; set; }
    public DateTimeOffset CreatedTime { get; set; } = DateTimeOffset.UtcNow;
    public Guid? UpdatedBy { get; set; }
    public DateTimeOffset? UpdatedTime { get; set; }
}

// ========== ValueObjects/ValueObject.cs ==========
namespace AKNexus.Framework.Domain.ValueObjects;

/// <summary>值对象基类</summary>
public abstract class ValueObject
{
    protected abstract IEnumerable<object?> GetAtomicValues();
    
    public override bool Equals(object? obj)
    {
        return obj is ValueObject other && GetAtomicValues().SequenceEqual(other.GetAtomicValues());
    }
    
    public override int GetHashCode()
    {
        return GetAtomicValues().Select(x => x?.GetHashCode() ?? 0).Aggregate((x, y) => x ^ y);
    }
}

// ========== Services/AppServiceBase.cs ==========
namespace AKNexus.Framework.Application.Services;

/// <summary>应用服务基类</summary>
public abstract class AppServiceBase<TEntity, TKey, TDto, TCreateInput, TUpdateInput, TRepository>
    where TEntity : class
    where TDto : class
    where TCreateInput : class
    where TUpdateInput : class
    where TRepository : IRepository<TEntity, TKey>
{
    protected readonly TRepository Repository;
    protected readonly IUnitOfWork UnitOfWork;
    protected readonly ILogger Logger;

    protected AppServiceBase(TRepository repository, IUnitOfWork unitOfWork, ILogger logger)
    {
        Repository = repository;
        UnitOfWork = unitOfWork;
        Logger = logger;
    }

    public virtual async Task<TDto?> GetByIdAsync(TKey id, CancellationToken ct = default)
    {
        var entity = await Repository.GetByIdAsync(id, ct);
        return entity?.Adapt<TDto>();  // 需要引用 Mapster
    }

    public virtual async Task<TDto> CreateAsync(TCreateInput input, CancellationToken ct = default)
    {
        var entity = input.Adapt<TEntity>();
        var result = await Repository.AddAsync(entity, ct);
        await UnitOfWork.SaveChangesAsync(ct);
        return result.Adapt<TDto>();
    }

    public virtual async Task DeleteAsync(TKey id, CancellationToken ct = default)
    {
        await Repository.DeleteByIdAsync(id, ct);
        await UnitOfWork.SaveChangesAsync(ct);
    }
}

// ========== Dtos/Result.cs ==========

// ========== Data/BaseDbContext.cs ==========
namespace AKNexus.Framework.Infrastructure.Data;

/// <summary>数据库上下文基类</summary>
public abstract class BaseDbContext : DbContext
{
    protected BaseDbContext(DbContextOptions options) : base(options) { }

    protected override void OnModelCreating(ModelBuilder modelBuilder)
    {
        base.OnModelCreating(modelBuilder);

        // 全局软删除过滤器
        foreach (var entityType in modelBuilder.Model.GetEntityTypes())
        {
            if (typeof(ISoftDelete).IsAssignableFrom(entityType.ClrType))
            {
                modelBuilder.Entity(entityType.ClrType)
                    .HasQueryFilter(e => EF.Property<bool>(e, "IsDeleted") == false);
            }
        }
    }

    public override int SaveChanges()
    {
        UpdateAuditEntities();
        return base.SaveChanges();
    }

    public override Task<int> SaveChangesAsync(CancellationToken ct = default)
    {
        UpdateAuditEntities();
        return base.SaveChangesAsync(ct);
    }

    /// <summary>自动更新审计字段</summary>
    private void UpdateAuditEntities()
    {
        foreach (var entry in ChangeTracker.Entries<IAuditable>())
        {
            if (entry.State == EntityState.Added)
                entry.Entity.CreatedTime = DateTimeOffset.UtcNow;
            else if (entry.State == EntityState.Modified)
                entry.Entity.UpdatedTime = DateTimeOffset.UtcNow;
        }

        foreach (var entry in ChangeTracker.Entries<ISoftDelete>())
        {
            if (entry.State == EntityState.Deleted)
            {
                entry.State = EntityState.Modified;
                entry.Entity.IsDeleted = true;
                entry.Entity.DeletedTime = DateTimeOffset.UtcNow;
            }
        }
    }
}

// ========== Data/DbContextFactory.cs ==========
namespace AKNexus.Framework.Infrastructure.Data;

/// <summary>数据库类型</summary>
public enum DatabaseType { PostgreSQL, SqlServer, MySql, Sqlite }

/// <summary>动态数据库切换配置</summary>
public class DatabaseConnectionConfig
{
    public DatabaseType DatabaseType { get; set; } = DatabaseType.PostgreSQL;
    public string ConnectionString { get; set; } = string.Empty;
}

/// <summary>数据库上下文工厂</summary>
public static class DbContextFactory
{
    public static IServiceCollection AddDynamicDbContext<TDbContext>(
        this IServiceCollection services,
        IConfiguration configuration,
        Action<DbContextOptionsBuilder, DatabaseType, string>? configure = null)
        where TDbContext : DbContext
    {
        services.AddDbContext<TDbContext>((sp, options) =>
        {
            var config = sp.GetRequiredService<DatabaseConnectionConfig>();
            var connectionString = configuration.GetConnectionString("Default") ?? config.ConnectionString;

            if (config.DatabaseType == DatabaseType.PostgreSQL)
                options.UseNpgsql(connectionString);
            // 其他数据库类型...
            
            configure?.Invoke(options, config.DatabaseType, connectionString);
        });

        services.AddScoped<DatabaseConnectionConfig>();
        return services;
    }
}

// ========== Repositories/EfRepositoryBase.cs ==========
namespace AKNexus.Framework.Infrastructure.Repositories;

/// <summary>EF Core 仓储基类</summary>
public abstract class EfRepositoryBase<TEntity, TKey, TDbContext> : IRepository<TEntity, TKey>
    where TEntity : class
    where TDbContext : DbContext
{
    protected readonly TDbContext DbContext;
    protected readonly DbSet<TEntity> DbSet;

    protected EfRepositoryBase(TDbContext dbContext)
    {
        DbContext = dbContext;
        DbSet = dbContext.Set<TEntity>();
    }

    public virtual async Task<TEntity?> GetByIdAsync(TKey id, CancellationToken ct = default)
        => await DbSet.FindAsync(new object[] { id! }, ct);

    public virtual async Task<IEnumerable<TEntity>> GetAllAsync(CancellationToken ct = default)
        => await DbSet.ToListAsync(ct);

    public virtual async Task<TEntity> AddAsync(TEntity entity, CancellationToken ct = default)
    {
        var entry = await DbSet.AddAsync(entity, ct);
        return entry.Entity;
    }

    public virtual async Task UpdateAsync(TEntity entity, CancellationToken ct = default)
    {
        DbSet.Update(entity);
        await Task.CompletedTask;
    }

    public virtual async Task DeleteAsync(TEntity entity, CancellationToken ct = default)
    {
        DbSet.Remove(entity);
        await Task.CompletedTask;
    }
}

// ========== UnitOfWork/UnitOfWork.cs ==========
namespace AKNexus.Framework.Infrastructure.UnitOfWork;

/// <summary>工作单元实现</summary>
public class UnitOfWork<TDbContext> : IUnitOfWork where TDbContext : DbContext
{
    private readonly TDbContext _dbContext;
    private IDbContextTransaction? _transaction;

    public UnitOfWork(TDbContext dbContext) => _dbContext = dbContext;

    public async Task<int> SaveChangesAsync(CancellationToken ct = default)
        => await _dbContext.SaveChangesAsync(ct);

    public async Task BeginTransactionAsync(CancellationToken ct = default)
        => _transaction = await _dbContext.Database.BeginTransactionAsync(ct);

    public async Task CommitTransactionAsync(CancellationToken ct = default)
    {
        await _dbContext.SaveChangesAsync(ct);
        if (_transaction != null) await _transaction.CommitAsync(ct);
    }

    public async Task RollbackTransactionAsync(CancellationToken ct = default)
    {
        if (_transaction != null) await _transaction.RollbackAsync(ct);
    }

    public void Dispose() => _transaction?.Dispose();
}

// ========== Controllers/ApiControllerBase.cs ==========
namespace AKNexus.Framework.Web.Controllers;

/// <summary>API 控制器基类</summary>
[ApiController]
[Route("api/[controller]")]
public abstract class ApiControllerBase : ControllerBase
{
    protected IActionResult Success<T>(T data, string message = "操作成功")
        => Ok(Result<T>.Ok(data, message));

    protected IActionResult Success(string message = "操作成功")
        => Ok(Result.Ok(message));

    protected IActionResult Fail(string message, string? errorCode = null)
        => BadRequest(Result.Fail(message, errorCode));

    protected IActionResult NotFoundResult(string message = "资源不存在")
        => NotFound(Result.Fail(message, "NOT_FOUND"));
}

// ========== Extensions/WebApplicationExtensions.cs ==========
namespace AKNexus.Framework.Web.Extensions;

/// <summary>Web 应用扩展</summary>
public static class WebApplicationExtensions
{
    /// <summary>配置 Furion 服务</summary>
    public static IServiceCollection AddFrameworkWeb(
        this IServiceCollection services,
        IConfiguration configuration,
        Action<IServiceCollection>? configure = null)
    {
        // 添加控制器并启用 Furion
        services.AddControllers().AddInject();

        // 配置 CORS
        services.AddCors(options =>
        {
            options.AddPolicy("AllowAll", policy =>
                policy.AllowAnyOrigin().AllowAnyMethod().AllowAnyHeader());
        });

        // 健康检查
        services.AddHealthChecks();

        // 自定义配置
        configure?.Invoke(services);

        return services;
    }

    /// <summary>配置 Web 中间件</summary>
    public static WebApplication UseFrameworkWeb(this WebApplication app)
    {
        // Furion 中间件（包含 Swagger）
        app.UseInject();

        // CORS
        app.UseCors("AllowAll");

        // 认证授权
        app.UseAuthentication();
        app.UseAuthorization();

        // 健康检查
        app.MapHealthChecks("/health");

        // 控制器路由
        app.MapControllers();

        return app;
    }
}

// ========== Middlewares/GlobalExceptionMiddleware.cs ==========
namespace AKNexus.Framework.Web.Middlewares;

/// <summary>全局异常中间件</summary>
public class GlobalExceptionMiddleware
{
    private readonly RequestDelegate _next;

    public GlobalExceptionMiddleware(RequestDelegate next) => _next = next;

    public async Task InvokeAsync(HttpContext context)
    {
        try
        {
            await _next(context);
        }
        catch (Exception ex)
        {
            await HandleExceptionAsync(context, ex);
        }
    }

    private static async Task HandleExceptionAsync(HttpContext context, Exception exception)
    {
        context.Response.ContentType = "application/json";
        context.Response.StatusCode = exception switch
        {
            ArgumentException => 400,
            UnauthorizedAccessException => 401,
            _ => 500
        };

        await context.Response.WriteAsJsonAsync(new
        {
            Success = false,
            Message = exception.Message,
            ErrorCode = context.Response.StatusCode.ToString()
        });
    }
}

## 三、主应用启动配置

```csharp
// ========== Program.cs ==========
using Furion;

var builder = WebApplication.CreateBuilder(args).Inject();

// 配置服务
services.AddFrameworkWeb(configuration, frameworkServices =>
{
    // 数据库
    frameworkServices.AddDynamicDbContext<HubDbContext>(configuration);
    
    // 仓储和工作单元
    frameworkServices.AddScoped(typeof(IRepository<>), typeof(Repository<>));
    frameworkServices.AddScoped<IUnitOfWork, UnitOfWork<HubDbContext>>();
    
    // 业务服务
    frameworkServices.AddScoped<IUserService, UserService>();
});

// JWT 认证
services.AddJwt<JwtHandler>();

var app = builder.Build();
app.UseFrameworkWeb();
app.Run();
```
